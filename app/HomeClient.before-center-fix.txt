function SiteLink({ href, children, className = '', ...props }) {
  const safe = safeHomeLink(href) || '/'

  if (/^https?:\/\//.test(safe)) {
    return (
      <a
        href={safe}
        target="_blank"
        rel="noopener noreferrer"
        className={className}
        {...props}
      >
        {children}
      </a>
    )
  }

  return (
    <Link href={safe} className={className} {...props}>
      {children}
    </Link>
  )
}

function SectionHeading({ eyebrow, title, body }) {

      <h2>{title}</h2>
      {body && <p className="sub">{body}</p>}
    </>
  )
}

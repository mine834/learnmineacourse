0416: const isObject = value =>
0417:   value !== null &&
0418:   typeof value === 'object' &&
0419:   !Array.isArray(value)
0420: 
0421: const text = value =>
0422:   typeof value === 'string' ? value.trim() : ''

0424: export function safeImageUrl(value) {
0425:   try {
0426:     const url = new URL(text(value))
0427:     return ['https:', 'http:'].includes(url.protocol) &&
0428:       !url.username &&
0429:       !url.password
0430:       ? url.href
0431:       : ''
0432:   } catch {
0433:     return ''
0434:   }
0435: }

0437: export function safeHomeLink(value) {
0438:   const link = text(value)
0439:   if (
0440:     link.startsWith('/') &&
0441:     !link.startsWith('//') &&
0442:     !link.includes('\\')
0443:   ) return link
0444: 
0445:   if (link.startsWith('#') && /^#[a-zA-Z0-9_-]+$/.test(link)) {
0446:     return link
0447:   }
0448: 
0449:   return safeImageUrl(link)
0450: }

0452: function mergeKnown(defaults, input) {
0453:   if (Array.isArray(defaults)) {
0454:     if (!Array.isArray(input)) return structuredClone(defaults)
0455: 
0456:     return input.slice(0, 30).map((item, index) => {
0457:       const template = defaults[index] ??
0458:         defaults[0] ??
0459:         (isObject(item) ? {} : '')
0460: 
0461:       return mergeKnown(template, item)
0462:     })
0463:   }
0464: 
0465:   if (isObject(defaults)) {
0466:     const result = {}
0467: 
0468:     for (const [key, value] of Object.entries(defaults)) {
0469:       result[key] = mergeKnown(value, isObject(input) ? input[key] : undefined)
0470:     }
0471: 
0472:     return result
0473:   }
0474: 
0475:   if (typeof defaults === 'boolean') {
0476:     return typeof input === 'boolean' ? input : defaults
0477:   }
0478: 
0479:   if (typeof defaults === 'number') {
0480:     const number = Number(input)
0481:     return input !== undefined && input !== null &&
0482:       input !== '' && Number.isFinite(number)
0483:       ? Math.max(0, Math.min(200, number))
0484:       : defaults
0485:   }
0486: 
0487:   return typeof input === 'string' ? input : defaults
0488: }

0490: export function normalizeHomeContent(input) {
0491:   const source = isObject(input) ? input : {}
0492:   const result = mergeKnown(homeContentDefaults, source)
0493: 
0494:   result.lifestyle.image_url = safeImageUrl(
0495:     result.lifestyle.image_url
0496:   )
0497: 
0498:   result.lifestyle.tags = result.lifestyle.tags
0499:     .map(text)
0500:     .filter(Boolean)
0501: 
0502:   result.registration.button_url = googleFormUrl(
0503:     result.registration.button_url
0504:   )
0505: 
0506:   for (const section of ['conversation', 'books', 'assessment', 'finalCta']) {
0507:     result[section].button_url =
0508:       safeHomeLink(result[section].button_url) ||
0509:       homeContentDefaults[section].button_url
0510:   }
0511: 
0512:   result.footer.instagram = safeImageUrl(result.footer.instagram)
0513:   result.footer.facebook = safeImageUrl(result.footer.facebook)
0514: 
0515:   const email = text(result.footer.email)
0516:   result.footer.email =
0517:     /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) ? email : ''
0518: 
0519:   result.testimonials.reviews = (
0520:     Array.isArray(source.testimonials?.reviews)
0521:       ? source.testimonials.reviews
0522:       : []
0523:   ).slice(0, 30).map((review, index) => ({
0524:     id: text(review?.id) || `review-${index}`,
0525:     name: text(review?.name),
0526:     text: text(review?.text),
0527:     caption: text(review?.caption),
0528:     published: review?.published === true,
0529:   }))
0530: 
0531:   for (const key of Object.keys(result.assessment.style)) {
0532:     const original = homeContentDefaults.assessment.style[key]
0533:     if (typeof original === 'string') {
0534:       const color = result.assessment.style[key]
0535:       result.assessment.style[key] =
0536:         /^#[0-9a-fA-F]{6}$/.test(color) ? color : original
0537:     }
0538:   }
0539: 
0540:   for (const language of ['EN', 'KR']) {
0541:     for (const section of ['study', 'conversation', 'books', 'assessment', 'finalCta']) {
0542:       const translated = result.translations[language]?.[section]
0543:       if (translated?.button_url) {
0544:         translated.button_url = safeHomeLink(translated.button_url)
0545:       }
0546:     }
0547:   }
0548: 
0549:   return result
0550: }

0556:     )
0557:   } catch {
0558:     return normalizeHomeContent(null)
0559:   }
